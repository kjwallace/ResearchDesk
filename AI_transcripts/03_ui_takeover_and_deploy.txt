    """'EPS', 'revenue', or a driver's label for a driver metric (given as an ID or a bare name)."""

def metric_value(metric: str, value: float | None, unit: str | None = None) -> str:
    """A projection figure with its unit: '12.5%', '$81.0bn', '$6.55'."""
